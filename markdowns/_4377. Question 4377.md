class BusBooking {
    private boolean[][] booked;
    private int[][] cost;
    private int totalTime;
    private int n;
    private int bookedCount = 0;

    public BusBooking(int n) {
        this.n = n;
        booked = new boolean[n][4];
        cost = new int[n][4];
        totalTime = 0;
        bookedCount = 0;
    }
    
    public void toggle(String seat) {
        int col = seat.charAt(0) - 'A';
        int row = Integer.parseInt(seat.substring(1)) - 1;

        if(booked[row][col]){
            totalTime -= cost[row][col];
            booked[row][col] = false;
            cost[row][col] = 0;
            bookedCount--;
        } else {
            int time = 1;

            if(col == 0 && booked[row][1]){
                time = 3;
            }else if(col == 3 && booked[row][2]){
                time = 3;
            }

            booked[row][col] = true;
            cost[row][col] = time;
            totalTime += time;
            bookedCount++;
        }
    }
    
    public int getTotalTime() {
        return totalTime;
    }
    
    public int getMinTime() {
        // int count = 0;

        // for(int i = 0; i < n; i++){
        //     for(int j = 0; j < 4; j++){
        //         if(booked[i][j]){
        //             count++;
        //         }
        //     }
        // }
        return bookedCount;
    }
}
