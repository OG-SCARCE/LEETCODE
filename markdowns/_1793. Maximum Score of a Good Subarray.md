#### [solution.java](<../solutions/1793-minimum-moves-to-make-array-complementary/solution.java>)
```Java
class Solution {

    public int minMoves(int[] nums, int limit) {

        int n = nums.length;

        // difference array
        int[] diff = new int[2 * limit + 2];

        for (int i = 0; i < n / 2; i++) {

            int a = nums[i];
            int b = nums[n - 1 - i];

            int low = Math.min(a, b) + 1;
            int high = Math.max(a, b) + limit;

            int sum = a + b;

            // initially all sums need 2 moves
            diff[2] += 2;
            diff[2 * limit + 1] -= 2;

            // one move range
            diff[low] -= 1;
            diff[high + 1] += 1;

            // exact sum needs 0 move
            diff[sum] -= 1;
            diff[sum + 1] += 1;
        }

        int ans = Integer.MAX_VALUE;
        int moves = 0;

        // calculate prefix sum
        for (int target = 2; target <= 2 * limit; target++) {
            moves += diff[target];
            ans = Math.min(ans, moves);
        }

        return ans;
    }
}
```
